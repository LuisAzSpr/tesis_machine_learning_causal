\begin{frame}{Problema: la correlación no responde la pregunta}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{itemize}
\item Pregunta contrafactual: \textbf{¿cuánto cambiaría el alquiler si el mismo departamento tuviera un baño más?}
\item Los departamentos con más baños son también más grandes, más nuevos y mejor ubicados.
\item Los estudios locales \textbf{predicen} precios (hedónico, XGBoost); no identifican efectos.
\item ML flexible $\neq$ causalidad: hace falta un modelo causal explícito.
\end{itemize}
\column{0.42\textwidth}
\vspace{0.8cm}
\begin{center}
\dato{+45\%}{asociación\\sin ajustar}\hspace{0.5em}\dato{?}{efecto\\causal}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{Alcance, objetivos e hipótesis}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Alcance}
\begin{itemize}\small
\item Precio \emph{de oferta} mensual, no contrato.
\item Avisos de un portal, 13 distritos, marzo 2025.
\item Tratamiento: nº de baños (1--5). Resultado: log(precio).
\end{itemize}
\textbf{Objetivos}
\begin{itemize}\small
\item OE1 auditoría · OE2 identificación · OE3 estimación
\item OE4 heterogeneidad y sensibilidad · OE6 simulación · OE7 trazabilidad
\item OE5 (IA sobre el texto): \textcolor{gris}{diferido}
\end{itemize}
\column{0.47\textwidth}
\textbf{Hipótesis} (sin dirección obligatoria)
\begin{itemize}\small
\item H1: el ajuste cambia la asociación.
\item H2: la composición causal de los controles importa más que el algoritmo.
\item H3: hay heterogeneidad, solo si la precisión la respalda.
\item H4: la solidez depende de la fuerza de los confusores no observados.
\end{itemize}
\end{columns}
\end{frame}

% 6 -----------------------------------------------------------------
\begin{frame}{Método: Double ML con dependencia espacial}
\begin{enumerate}
\item Predecir el precio y los baños a partir de los confusores con \textit{gradient boosting} (relaciones no lineales).
\item \textit{Cross-fitting}: cada aviso se predice con modelos entrenados \textbf{sin su edificio} (5 folds agrupados $\times$ 10 repeticiones).
\item Residuos: $\tilde Y=Y-\hat E[Y|X]$, \ $\tilde T=T-\hat E[T|X]$.
\item Efecto: pendiente de $\tilde Y$ sobre $\tilde T$.
\item Inferencia \textbf{agrupada por edificio} (alternativa: bloques de 1 km).
\end{enumerate}
\vspace{0.3cm}
\small Además: grilla preespecificada de controles, sensibilidad a confusores omitidos, efectos por grupos (GATE), I de Moran y validación semisintética.
\end{frame}

\begin{frame}{Robustez a confusores no observados (R5)}
\begin{columns}[c]
\column{0.5\textwidth}
\includegraphics[width=\textwidth]{F5_sensibilidad.png}
\column{0.48\textwidth}\small
\begin{itemize}
\item \textbf{RV = 0.075}: un $U$ que explique el 7.5\,\% de la varianza residual de los baños \emph{y} del precio anula el efecto.
\item $U$ tan fuerte como la \textbf{antigüedad} $\rightarrow$ 3.6\,\% (sigue siendo $>0$).
\item $U$ tan fuerte como la \textbf{ubicación} $\rightarrow$ 2.5\,\% (el IC incluye 0).
\item $U$ tan fuerte como el \textbf{área} $\rightarrow$ cambia de signo.
\end{itemize}
\textbf{H4 se confirma}: fragilidad \textbf{moderada}.
\end{columns}
\end{frame}

\begin{frame}{Heterogeneidad y dependencia espacial (R7)}
\centering\includegraphics[height=0.5\textheight]{F7_heterogeneidad.png}\par
\small
\begin{itemize}
\item Sin diferencias significativas por clase de distrito ni por tamaño (Wald $p>0.6$) $\rightarrow$ \textbf{H3 sin respaldo} (no concluyente: baja potencia).
\item Residuos del precio con autocorrelación espacial (I de Moran = 0.06, $p=0.001$): EE agrupados 31--39\,\% mayores que los iid; el IC sigue excluyendo el 0.
\end{itemize}
\end{frame}

% 11 ----------------------------------------------------------------
\begin{frame}{Validación semisintética: ¿funciona con verdad conocida? (R6)}
\centering\includegraphics[height=0.5\textheight]{F6_semisintetico.png}\par
\footnotesize
\begin{itemize}
\item Covariables reales, $\theta_0=0.05$, 600 réplicas. \textbf{Sin $U$}: DML insesgado y cobertura 95\,\%; el hedónico lineal se sesga (+0.025; cobertura 50\,\%).
\item \textbf{Con $U$}: el DML se sesga y el IC no lo detecta; la \textbf{cota de sensibilidad sí} recupera la cobertura (92--96\,\%).
\item Heterogeneidad real ($0.03$ vs $0.08$) detectada solo el 24\,\% de las veces. Dependencia espacial: hay que agrupar al nivel correcto.
\end{itemize}
\end{frame}

\begin{itemize}\small

\item Requiere: taxonomía, 200--300 avisos etiquetados a mano, precisión/\textit{recall}, posición causal en el DAG.

\end{itemize}
\column{0.48\textwidth}
\textbf{Límites}
\begin{itemize}\small
\item Ignorabilidad: se acota, no se prueba.
\item Un portal, un mes, precio de oferta.
\item Antigüedad faltante en el 25\,\%.
\item Efecto lineal por baño; sin diagnóstico formal de soporte común.
\end{itemize}

% 13 ----------------------------------------------------------------
\begin{frame}{Conclusiones}
\begin{itemize}
\item \textbf{Respuesta}: bajo los supuestos del DAG, un baño adicional eleva el precio de oferta en \textbf{5.3\,\%} (1.9--8.8\,\%), unos S/\,133 al mes.
\item \textbf{Hallazgo observacional}: el 86\,\% de la asociación bruta se explica por tamaño, antigüedad y ubicación.
\item \textbf{Metodológico}: el hedónico lineal sobreestima incluso sin confusión omitida; los malos controles destruyen la estimación.
\item \textbf{Fragilidad}: moderada; resiste un confusor como la antigüedad, no como la ubicación.
\item \textbf{No se afirma}: que construir un baño sea rentable, ni que el efecto valga para contratos o para toda Lima.
\end{itemize}
\end{frame}
